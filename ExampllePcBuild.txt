local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local CoreGui = game:GetService("CoreGui")
local Camera = workspace.CurrentCamera
local LocalPlayer = Players.LocalPlayer

local Library = loadstring(game:HttpGet("https://raw.githubusercontent.com/Essluau/EssnceLibrary/refs/heads/main/EssncePcBuild.lua"))()

local Window = Library:CreateWindow({
	Name = "EssncePw",
	Accent = Color3.fromHex("#CF00FF"),
	ConfigFolder = "EssncePw_Configs"
})

local CombatTab = Window:CreateTab("Combat")

local CombatSub1 = CombatTab:CreateSubTab({
	Icon = "https://raw.githubusercontent.com/Essluau/icons/refs/heads/main/Crosshair.png",
	IconFileName = "combat_1.png",
	IconText = "❖",
	ActiveSize = 24,
	InactiveSize = 20
})

local CombatSub2 = CombatTab:CreateSubTab({
	Icon = "https://raw.githubusercontent.com/Essluau/icons/refs/heads/main/add_moderator.png",
	IconFileName = "combat_2.png",
	IconText = "☁",
	ActiveSize = 24,
	InactiveSize = 20
})

local VisualTab = Window:CreateTab("Visual")

local VisualSub1 = VisualTab:CreateSubTab({
	Icon = "https://raw.githubusercontent.com/Essluau/icons/refs/heads/main/ar_on_you__.png",
	IconFileName = "visuals_1.png",
	IconText = "👁",
	ActiveSize = 24,
	InactiveSize = 20
})

local EspCard = VisualSub1:CreateCard("Left", "ESP Controls", 480)

local espToggle = EspCard:AddToggle({
	Name = "Box ESP",
	Default = false,
	Type = "dual_colorbox",
	Callback = function(state)
		print("Box ESP state:", state)
	end
})

EspCard:AddKeybind({
	Name = "ESP Keybind",
	Default = Enum.KeyCode.V,
	Callback = function()
		espToggle.Set(not espToggle.Get())
	end
})

EspCard:AddColorPicker({
	Name = "Accent Override",
	Default = Color3.fromRGB(207, 0, 255),
	Callback = function(color)
		print("Picked color:", color)
	end
})

EspCard:AddSlider({
	Name = "Line Thickness",
	Default = 1.0,
	Min = 0.5,
	Max = 5.0,
	Precise = 1,
	Callback = function(val)
		print("Thickness slider:", val)
	end
})

EspCard:AddDropdown({
	Name = "Box Style",
	Default = "Default",
	Options = {"Default", "Corner", "Rounded", "Filled"},
	ZIndex = 35,
	Callback = function(selected)
		print("Dropdown selected:", selected)
	end
})

EspCard:AddMultiDropdown({
	Name = "ESP Overlays",
	Default = {"Boxes", "Names"},
	Options = {"Boxes", "Names", "Skeletons", "Chams", "Distance", "Health"},
	ZIndex = 30,
	Callback = function(selectedList)
		print("MultiDropdown selected:", table.concat(selectedList, ", "))
	end
})

EspCard:AddInput({
	Name = "Tag Label",
	Placeholder = "Enter player tag...",
	Default = "Player",
	Callback = function(text, enterPressed)
		print("Input submitted:", text, "Enter pressed:", enterPressed)
	end
})

EspCard:AddDualButtons({
	Text1 = "Reset",
	Callback1 = function()
		print("Reset clicked")
	end,
	Text2 = "Print Stats",
	Callback2 = function()
		print("Print Stats clicked")
	end
})

SkinsSub1:CreatePlayerList("Left", {
	Title = "Player List",
	SizeY = 360,
	TeamFilter = "All",
	Elements = {
		{
			type = "toggle",
			name = "Aim",
			default = false,
			callback = function(plr, state)
				print("[PlayerList] Aim for", plr.Name, ":", state)
			end
		},
		{
			type = "toggle",
			name = "ESP",
			default = true,
			callback = function(plr, state)
				print("[PlayerList] ESP for", plr.Name, ":", state)
			end
		},
		{
			type = "dropdown",
			name = "Prio",
			default = "Med",
			options = {"Low", "Med", "High"},
			callback = function(plr, selectedOption)
				print("[PlayerList] Priority for", plr.Name, ":", selectedOption)
			end
		}
	}
})
