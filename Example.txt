local Players = game:GetService("Players")
local RunService = game:GetService("RunService")

local Camera = workspace.CurrentCamera
local LocalPlayer = Players.LocalPlayer

local Library = loadstring(game:HttpGet("https://raw.githubusercontent.com/Essluau/EssnceLibrary/refs/heads/main/EssnceLibraryMobile.lua"))()

local EspConfig = {
	Enabled = false,
	Color = Color3.fromRGB(255, 0, 0),
	Thickness = 1.0,
	CornerType = "Sharp",
	AppliedText = "Player"
}

local tempCustomText = "Player"

local EspFolder = Instance.new("ScreenGui")
EspFolder.Name = "RenderEspContainer"
EspFolder.ResetOnSpawn = false
EspFolder.DisplayOrder = 999999
pcall(function()
	EspFolder.Parent = (gethui and gethui()) or CoreGui
end)
if not EspFolder.Parent then
	EspFolder.Parent = LocalPlayer:WaitForChild("PlayerGui")
end

local EspFrames = {}

local function applyCornerStyle(cornerInst, style)
	if style == "Sharp" then
		cornerInst.CornerRadius = UDim.new(0, 0)
	elseif style == "Slight" then
		cornerInst.CornerRadius = UDim.new(0, 4)
	elseif style == "Rounded" then
		cornerInst.CornerRadius = UDim.new(0, 10)
	end
end

local function createBoxUI(plr)
	if plr == LocalPlayer or EspFrames[plr] then return end

	local boxFrame = Instance.new("Frame")
	boxFrame.Name = "Box_" .. plr.UserId
	boxFrame.BackgroundTransparency = 1
	boxFrame.BorderSizePixel = 0
	boxFrame.Visible = false
	boxFrame.Parent = EspFolder

	local stroke = Instance.new("UIStroke")
	stroke.Thickness = EspConfig.Thickness
	stroke.Color = EspConfig.Color
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = boxFrame

	local corner = Instance.new("UICorner")
	applyCornerStyle(corner, EspConfig.CornerType)
	corner.Parent = boxFrame

	local topLabel = Instance.new("TextLabel")
	topLabel.Name = "CustomText"
	topLabel.BackgroundTransparency = 1
	topLabel.Size = UDim2.new(1, 40, 0, 16)
	topLabel.Position = UDim2.new(0.5, -20, 0, -18)
	topLabel.AnchorPoint = Vector2.new(0.5, 0)
	topLabel.Font = Enum.Font.SourceSansBold
	topLabel.TextSize = 13
	topLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
	topLabel.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
	topLabel.TextStrokeTransparency = 0.2
	topLabel.TextXAlignment = Enum.TextXAlignment.Center
	topLabel.Text = EspConfig.AppliedText
	topLabel.Parent = boxFrame

	EspFrames[plr] = {
		Frame = boxFrame,
		Stroke = stroke,
		Corner = corner,
		Label = topLabel
	}
end

local function removeBoxUI(plr)
	if EspFrames[plr] then
		if EspFrames[plr].Frame then
			EspFrames[plr].Frame:Destroy()
		end
		EspFrames[plr] = nil
	end
end

for _, p in ipairs(Players:GetPlayers()) do
	createBoxUI(p)
end
Players.PlayerAdded:Connect(createBoxUI)
Players.PlayerRemoving:Connect(removeBoxUI)

RunService.RenderStepped:Connect(function()
	if not EspConfig.Enabled then
		for _, data in pairs(EspFrames) do
			if data.Frame.Visible then
				data.Frame.Visible = false
			end
		end
		return
	end

	for plr, data in pairs(EspFrames) do
		local char = plr.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChildOfClass("Humanoid")

		if hrp and hum and hum.Health > 0 then
			local rootPos, onScreen = Camera:WorldToViewportPoint(hrp.Position)

			if onScreen then
				local topPos = Camera:WorldToViewportPoint(hrp.Position + Vector3.new(0, 2.5, 0))
				local bottomPos = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3.2, 0))

				local height = math.abs(topPos.Y - bottomPos.Y)
				local width = height * 0.65

				data.Frame.Size = UDim2.new(0, width, 0, height)
				data.Frame.Position = UDim2.new(0, rootPos.X - (width / 2), 0, topPos.Y)
				data.Stroke.Color = EspConfig.Color
				data.Stroke.Thickness = EspConfig.Thickness
				data.Label.Text = EspConfig.AppliedText
				applyCornerStyle(data.Corner, EspConfig.CornerType)
				data.Frame.Visible = true
			else
				data.Frame.Visible = false
			end
		else
			data.Frame.Visible = false
		end
	end
end)
